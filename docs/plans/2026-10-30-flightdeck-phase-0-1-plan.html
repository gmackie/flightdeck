Goal

Events from a Worker, a Durable Object, and a browser page using data-flightdeck reach Flightdeck over service binding RPC, show up in live per-hour Durable Object counters, and land in Iceberg tables on R2.

Architecture

One flightdeck Worker: an Ingest WorkerEntrypoint fans out to Basin Pipelines, a SessionDO per session, and AppDO shards per app-hour. D1 holds the app registry. SDKs live in packages/.

Stack

pnpm 12, Node 24, TypeScript 7, Biome 2.6, Vitest 5 + @cloudflare/vitest-plugin, happy-dom, Valibot, jose, ulid, wrangler 4.156.

    const ID_SEGMENT = [
  /^\d+$/,
  /^[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}$/i,
  /^[0-9A-HJKMNP-TV-Z]{26}$/,

];

/** Collapses ids in a path so `/users/123` and `/users/456` become one journey step. */
export function normalizeRoute(path: string): string {
  const pathname = path.split(/[?#]/, 1)[0] || "/";
  const out = pathname
    .split("/")
    .map((seg) => (seg && ID_SEGMENT.some((re) => re.test(seg)) ? ":id" : seg))
    .join("/");
  return out.length > 1 && out.endsWith("/") ? out.slice(0, -1) : out || "/";
}
    packages/schema/src/user-agent.ts
    export type UserAgentInfo = {
  device: "mobile" | "tablet" | "desktop" | "bot" | "unknown";
  browser: string;
  os: string;
};

/** Coarse device, browser, and OS families. Deliberately small; no full UA database. */
export function parseUserAgent(ua: string | undefined): UserAgentInfo {
  if (!ua) return { device: "unknown", browser: "Other", os: "Other" };
  const bot = /bot|crawl|spider|slurp|headless|lighthouse/i.test(ua);
  const tablet = /iPad|Tablet/i.test(ua) || (/Android/.test(ua) && !/Mobile/.test(ua));
  const mobile = !tablet && /Mobi|iPhone|iPod|Android/.test(ua);
  const browser = /Edg\//.test(ua)
    ? "Edge"
    : /OPR\/|Opera/.test(ua)
      ? "Opera"
      : /Firefox\/|FxiOS/.test(ua)
        ? "Firefox"
        : /Chrome\/|CriOS/.test(ua)
          ? "Chrome"
          : /Safari\//.test(ua)
            ? "Safari"
            : "Other";
  const os = /iPhone|iPad|iPod/.test(ua)
    ? "iOS"
    : /Android/.test(ua)
      ? "Android"
      : /CrOS/.test(ua)
        ? "ChromeOS"
        : /Mac OS X|Macintosh/.test(ua)
          ? "macOS"
          : /Windows/.test(ua)
            ? "Windows"
            : /Linux/.test(ua)
              ? "Linux"
              : "Other";
  return { device: bot ? "bot" : tablet ? "tablet" : mobile ? "mobile" : "desktop", browser, os };
}

import * as v from "valibot";

const MAX_FUTURE_MS = 5 * 60_000;
const MAX_PAST_MS = 24 * 60 * 60_000;

export interface IngestDeps {
  now(): number;
  newId(): string;
  resolveUser(accessJwt: string): Promise<string | null>;
  sendEvents(rows: StoredEvent[]): Promise<void>;
  appendToSession(appId: string, sessionId: string, rows: StoredEvent[]): Promise<void>;
  recordCounts(appId: string, hour: string, rows: StoredEvent[]): Promise<void>;
}

export class IngestError extends Error {
  constructor(message: string, readonly issues?: unknown) {
    super(message);
  }
}

export async function ingestBatch(appId: string, input: unknown, source: EventSource, deps: IngestDeps): Promise<IngestResult> {
  const parsed = v.safeParse(IngestBatch, input);
  if (!parsed.success) throw new IngestError("invalid batch", v.flatten(parsed.issues));
  const { events, context, accessJwt } = parsed.output;

  const ua = parseUserAgent(context?.userAgent);
  if (ua.device === "bot") return { accepted: 0 };
  const receivedAt = deps.now();
  const accessUser = accessJwt ? await deps.resolveUser(accessJwt) : null;
  const rows = events.map((e) =>
    toRow(appId, e, { id: deps.newId(), receivedAt, ua, accessUser, source, country: context?.country ?? null }),
  );

  const work = [deps.sendEvents(rows)];
  for (const [sessionId, group] of groupBy(rows, (r) => r.session_id)) {
    work.push(deps.appendToSession(appId, sessionId, group));
  }
  for (const [hour, group] of groupBy(rows, (r) => hourKey(r.ts))) {
    work.push(deps.recordCounts(appId, hour, group));
  }
  await Promise.all(work);
  return { accepted: rows.length };
}

type RowContext = {
  id: string;
  receivedAt: number;
  ua: UserAgentInfo;
  accessUser: string | null;
  source: EventSource;
  country: string | null;
};

function toRow(appId: string, e: ClientEvent, c: RowContext): StoredEvent {
  const skewed = e.ts > c.receivedAt + MAX_FUTURE_MS || e.ts < c.receivedAt - MAX_PAST_MS;
  return {
    event_id: c.id,
    app_id: appId,
    kind: e.kind,
    name: e.name,
    ts: skewed ? c.receivedAt : e.ts,
    received_at: c.receivedAt,
    session_id: e.sessionId ?? null,
    anonymous_id: e.anonymousId ?? null,
    user_id: e.userId ?? c.accessUser,
    path: e.path ? normalizeRoute(e.path) : null,
    referrer: e.referrer ? referrerHost(e.referrer) : null,
    props: e.props ? JSON.stringify(e.props) : null,
    country: c.country,
    device: c.ua.device,
    browser: c.ua.browser,
    os: c.ua.os,
    source: c.source,
  };
}

/** Keep only the referring host; full referrer URLs often carry personal data. */
function referrerHost(referrer: string): string | null {
  try {
    return new URL(referrer).host || null;
  } catch {
    return null;
  }
}

function groupBy<T>(items: T[], key: (item: T) => string | null): Map<string, T[]> {
  const groups = new Map<string, T[]>();
  for (const item of items) {
    const k = key(item);
    if (k === null) continue;
    const group = groups.get(k);
    if (group) group.push(item);
    else groups.set(k, [item]);
  }
  return groups;
}

/** Live counters for one app and one UTC hour. */
export class AppDO extends DurableObject<Env> {
  private readonly sql: SqlStorage;

  constructor(ctx: DurableObjectState, env: Env) {
    super(ctx, env);
    this.sql = ctx.storage.sql;
    this.sql.exec(`
      CREATE TABLE IF NOT EXISTS counters (kind TEXT NOT NULL, name TEXT NOT NULL, count INTEGER NOT NULL, PRIMARY KEY (kind, name));
      CREATE TABLE IF NOT EXISTS transitions (source TEXT NOT NULL, target TEXT NOT NULL, count INTEGER NOT NULL, PRIMARY KEY (source, target));
      CREATE TABLE IF NOT EXISTS sessions (session_id TEXT PRIMARY KEY);
    `);
  }

  recordEvents(rows: StoredEvent[]): void {
    for (const r of rows) {
      this.sql.exec(
        "INSERT INTO counters (kind, name, count) VALUES (?, ?, 1) ON CONFLICT (kind, name) DO UPDATE SET count = count + 1",
        r.kind,
        r.kind === "page" ? (r.path ?? r.name) : r.name,
      );
      if (r.session_id) this.sql.exec("INSERT OR IGNORE INTO sessions (session_id) VALUES (?)", r.session_id);
    }
  }

  recordTransitions(transitions: Transition[]): void {
    for (const t of transitions) {
      this.sql.exec(
        "INSERT INTO transitions (source, target, count) VALUES (?, ?, ?) ON CONFLICT (source, target) DO UPDATE SET count = count + excluded.count",
        t.source,
        t.target,
        t.count,
      );
    }
  }

  snapshot(): AppSnapshot {
    return {
      counters: this.sql.exec<{ kind: string; name: string; count: number }>("SELECT kind, name, count FROM counters ORDER BY kind, name").toArray(),
      transitions: this.sql.exec<Transition>("SELECT source, target, count FROM transitions ORDER BY source, target").toArray(),
      sessions: this.sql.exec<{ n: number }>("SELECT count(*) AS n FROM sessions").one().n,
    };
  }

import { DurableObject } from "cloudflare:workers";
import { sinks } from "../sinks";
import { appShard } from "./shards";

const IDLE_MS = 30 * 60_000;
const MAX_STEPS = 200;

type SessionState = {
  appId: string;
  sessionId: string;
  anonymousId: string | null;
  userId: string | null;
  startedAt: number;
  lastTs: number;
  steps: string[];
  eventCount: number;
  country: string | null;
  device: string;
};

/** One visitor session: orders its steps, streams transitions, and writes a path record when idle. */
export class SessionDO extends DurableObject<Env> {
  async append(appId: string, sessionId: string, rows: StoredEvent[]): Promise<void> {
    const sorted = [...rows].sort((a, b) => a.ts - b.ts);
    const first = sorted[0];
    if (!first) return;
    const state = (await this.ctx.storage.get<SessionState>("state")) ?? {
      appId,
      sessionId,
      anonymousId: null,
      userId: null,
      startedAt: first.ts,
      lastTs: first.ts,
      steps: [],
      eventCount: 0,
      country: first.country,
      device: first.device,
    };

    const byHour = new Map<string, Map<string, Transition>>();
    for (const r of sorted) {
      state.eventCount++;
      state.lastTs = Math.max(state.lastTs, r.ts);
      state.userId = r.user_id ?? state.userId;
      state.anonymousId ??= r.anonymous_id;
      const step = stepName(r);
      const prev = state.steps.at(-1);
      if (!step || step === prev || state.steps.length >= MAX_STEPS) continue;
      addTransition(byHour, hourKey(r.ts), prev ?? START, step);
      state.steps.push(step);
    }

    await this.ctx.storage.put("state", state);
    await this.ctx.storage.setAlarm(Date.now() + IDLE_MS);
    await Promise.all(
      [...byHour].map(([hour, ts]) => appShard(this.env, appId, hour).recordTransitions([...ts.values()])),
    );
  }

  async alarm(): Promise<void> {
    const state = await this.ctx.storage.get<SessionState>("state");
    if (!state) return;
    await sinks.sessions(this.env, [toRecord(state)]);
    const exit = state.steps.at(-1);
    if (exit) {
      await appShard(this.env, state.appId, hourKey(state.lastTs)).recordTransitions([{ source: exit, target: EXIT, count: 1 }]);
    }
    await this.ctx.storage.deleteAll();
  }
}

function addTransition(byHour: Map<string, Map<string, Transition>>, hour: string, source: string, target: string) {
  let hourMap = byHour.get(hour);
  if (!hourMap) byHour.set(hour, (hourMap = new Map()));
  const key = `${source}\u0000${target}`;
  const t = hourMap.get(key);
  if (t) t.count++;
  else hourMap.set(key, { source, target, count: 1 });
}

function toRecord(s: SessionState): SessionRecord {
  return {
    session_id: s.sessionId,
    app_id: s.appId,
    anonymous_id: s.anonymousId,
    user_id: s.userId,
    started_at: s.startedAt,
    ended_at: s.lastTs,
    duration_ms: s.lastTs - s.startedAt,
    entry: s.steps[0] ?? null,
    exit: s.steps.at(-1) ?? null,
    steps: JSON.stringify(s.steps),
    step_count: s.steps.length,
    event_count: s.eventCount,
    country: s.country,
    device: s.device,
  };
}

    import type { IngestRpc, RequestContext } from "@flightdeckhq/schema";

export type { IngestResult, IngestRpc } from "@flightdeckhq/schema";

export type FlightdeckOptions = {
  /** Env binding name of the Flightdeck service binding. Default `FLIGHTDECK`. */
  binding?: string;
  /** Endpoint path the browser script posts to. Default `/_fd/e`. */
  path?: string;
  /** Extra origins allowed to post, besides the request's own origin. */
  allowedOrigins?: string[];
  /** Default 64 KiB. */
  maxBodyBytes?: number;
};

const DEFAULT_PATH = "/_fd/e";
const DEFAULT_MAX_BODY = 64 * 1024;

/** Handles the browser endpoint. Returns null when the request is not for Flightdeck. */
export async function handleFlightdeck(request: Request, env: object, options: FlightdeckOptions = {}): Promise<Response | null> {
  const url = new URL(request.url);
  if (url.pathname !== (options.path ?? DEFAULT_PATH)) return null;
  if (request.method !== "POST") return new Response(null, { status: 405, headers: { Allow: "POST" } });

  const origin = request.headers.get("Origin");
  if (origin && origin !== url.origin && !options.allowedOrigins?.includes(origin)) {
    return new Response(null, { status: 403 });
  }

  const bindingName = options.binding ?? "FLIGHTDECK";
  const binding = (env as Record<string, IngestRpc | undefined>)[bindingName];
  if (!binding) throw new Error(`Flightdeck: env.${bindingName} service binding is missing`);

  const text = await request.text();
  if (text.length > (options.maxBodyBytes ?? DEFAULT_MAX_BODY)) return new Response(null, { status: 413 });
  let body: unknown;
  try {
    body = JSON.parse(text);
  } catch {
    return new Response(null, { status: 400 });
  }
  if (typeof body !== "object" || body === null) return new Response(null, { status: 400 });

  const country = (request as { cf?: { country?: string } }).cf?.country;
  const userAgent = request.headers.get("User-Agent")?.slice(0, 512);
  const context: RequestContext = { ...(country ? { country } : {}), ...(userAgent ? { userAgent } : {}) };
  const accessJwt = request.headers.get("Cf-Access-Jwt-Assertion");

  try {
    await binding.ingest({ events: (body as { events?: unknown }).events, context, ...(accessJwt ? { accessJwt } : {}) });
  } catch (err) {
    const invalid = err instanceof Error && err.message.startsWith("invalid batch");
    if (!invalid) console.error("Flightdeck ingest failed", err);
    return new Response(null, { status: invalid ? 400 : 502 });
  }
  return new Response(null, { status: 204 });
}

/** Wraps a Worker's default export so it serves the Flightdeck browser endpoint. */
export function withFlightdeck<E extends object>(handler: ExportedHandler<E>, options?: FlightdeckOptions): ExportedHandler<E> {
  return {
    ...handler,
    async fetch(request, env, ctx) {
      const res = await handleFlightdeck(request as unknown as Request, env, options);
      if (res) return res;
      if (!handler.fetch) return new Response("Not found", { status: 404 });
      return handler.fetch(request, env, ctx);
    },
  };
}

export const SESSION_IDLE_MS = 30 * 60_000;

const memory = new Map<string, string>();
let ephemeralAnon: string | undefined;

export function randomId(): string {
  return crypto.randomUUID().replaceAll("-", "");
}

function read(store: () => Storage, key: string): string | null {
  try {
    return store().getItem(key);
  } catch {
    return memory.get(key) ?? null;
  }
}

function write(store: () => Storage, key: string, value: string): void {
  try {
    store().setItem(key, value);
  } catch {
    memory.set(key, value);
  }
}

/** Session id in sessionStorage, rotated after 30 minutes without activity. */
export function sessionId(now = Date.now()): string {
  const session = () => sessionStorage;
  let id = read(session, SESSION_KEY);
  if (!id || now - Number(read(session, LAST_KEY) ?? 0) > SESSION_IDLE_MS) {
    id = randomId();
    write(session, SESSION_KEY, id);
  }
  write(session, LAST_KEY, String(now));
  return id;
}

/** Persistent anonymous id in localStorage, or a page-lifetime id when persistence is off. */
export function anonymousId(persist: boolean): string {
  if (!persist) return (ephemeralAnon ??= randomId());
  const local = () => localStorage;
  let id = read(local, ANON_KEY);
  if (!id) {
    id = randomId();
    write(local, ANON_KEY, id);
  }
  return id;
}

    import type { ClientEvent } from "@flightdeckhq/schema";

const MAX_BATCH = 50;

export type Transport = (endpoint: string, body: string) => void;

export function sendBatch(endpoint: string, body: string): void {
  if (navigator.sendBeacon?.(endpoint, new Blob([body], { type: "text/plain" }))) return;
  void fetch(endpoint, { method: "POST", body, keepalive: true, headers: { "Content-Type": "text/plain" } }).catch(() => {});
}

export class EventQueue {
  private events: ClientEvent[] = [];

  constructor(
    private readonly opts: { endpoint: string; maxBatch: number },
    private readonly transport: Transport = sendBatch,
  ) {}

  push(event: ClientEvent): void {
    this.events.push(event);
    if (this.events.length >= this.opts.maxBatch) this.flush();
  }

  flush(): void {
    while (this.events.length) {
      this.transport(this.opts.endpoint, JSON.stringify({ events: this.events.splice(0, MAX_BATCH) }));
    }
  }
}

    import type { EventKind } from "@flightdeckhq/schema";

export const ATTR = "data-flightdeck";
const VIEW_ATTR = `${ATTR}-view`;

export type Emit = (kind: EventKind, name: string, props?: Record<string, string>) => void;

/** `data-flightdeck-plan="pro"` becomes `{ plan: "pro" }`. */
export function propsFrom(el: HTMLElement): Record<string, string> | undefined {
  const props: Record<string, string> = {};
  for (const [key, value] of Object.entries(el.dataset)) {
    if (!key.startsWith("flightdeck") || key === "flightdeck" || key === "flightdeckView" || value === undefined) continue;
    const name = key.slice("flightdeck".length);
    props[name.charAt(0).toLowerCase() + name.slice(1)] = value;
  }
  return Object.keys(props).length ? props : undefined;
}

export function installCapture(doc: Document, emit: Emit): () => void {
  const onClick = (ev: Event) => {
    const el = (ev.target as Element | null)?.closest?.<HTMLElement>(`[${ATTR}]`);
    if (!el || el.tagName === "FORM" || el.hasAttribute(VIEW_ATTR)) return;
    emit("click", el.getAttribute(ATTR) ?? "", propsFrom(el));
  };
  const onSubmit = (ev: Event) => {
    const form = ev.target as HTMLElement;
    const name = form.getAttribute?.(ATTR);
    if (name) emit("submit", name, propsFrom(form));
  };
  doc.addEventListener("click", onClick, true);
  doc.addEventListener("submit", onSubmit, true);

  const seen = new WeakSet<Element>();
  const io =
    typeof IntersectionObserver === "undefined"
      ? null
      : new IntersectionObserver(
          (entries) => {
            for (const entry of entries) {
              if (!entry.isIntersecting || seen.has(entry.target)) continue;
              seen.add(entry.target);
              io?.unobserve(entry.target);
              const el = entry.target as HTMLElement;
              emit("view", el.getAttribute(ATTR) ?? "", propsFrom(el));
            }
          },
          { threshold: 0.5 },
        );
  const observeAll = () => {
    for (const el of doc.querySelectorAll(`[${ATTR}][${VIEW_ATTR}]`)) if (!seen.has(el)) io?.observe(el);
  };
  observeAll();
  const mo = new MutationObserver(observeAll);
  mo.observe(doc.documentElement, { childList: true, subtree: true });

  return () => {
    doc.removeEventListener("click", onClick, true);
    doc.removeEventListener("submit", onSubmit, true);
    io?.disconnect();
    mo.disconnect();
  };
}

export type InitOptions = {
  /** Same-origin endpoint served by withFlightdeck/handleFlightdeck. Default `/_fd/e`. */
  endpoint?: string;
  /** Keep a persistent anonymous id in localStorage. Default true. */
  persistAnonymousId?: boolean;

  trackPages?: boolean;
  maxBatch?: number;
  flushIntervalMs?: number;
};

class Client {
  private readonly queue: EventQueue;
  private readonly persist: boolean;
  private readonly cleanup: (() => void)[] = [];
  private userId: string | undefined;
  private lastPath: string | undefined;

  constructor(o: InitOptions) {
    this.persist = o.persistAnonymousId ?? true;
    this.queue = new EventQueue({ endpoint: o.endpoint ?? "/_fd/e", maxBatch: o.maxBatch ?? 20 });
    const flush = () => this.queue.flush();
    const onHidden = () => document.visibilityState === "hidden" && flush();
    const timer = setInterval(flush, o.flushIntervalMs ?? 5_000);
    document.addEventListener("visibilitychange", onHidden);
    addEventListener("pagehide", flush);
    this.cleanup.push(() => {
      clearInterval(timer);
      document.removeEventListener("visibilitychange", onHidden);
      removeEventListener("pagehide", flush);
    });
    this.cleanup.push(installCapture(document, (kind, name, props) => this.emit(kind, name, props)));
    if (o.trackPages ?? true) {
      this.page();
      this.cleanup.push(watchHistory(() => this.page()));
    }
  }

  emit(kind: EventKind, name: string, props?: Record<string, string | number | boolean | null>): void {
    this.queue.push({
      kind,
      name,
      ts: Date.now(),
      sessionId: sessionId(),
      anonymousId: anonymousId(this.persist),
      path: location.pathname,
      ...(this.userId ? { userId: this.userId } : {}),
      ...(props ? { props } : {}),
    });
  }

  page(): void {
    if (location.pathname === this.lastPath) return;
    const first = this.lastPath === undefined;
    this.lastPath = location.pathname;
    this.queue.push({
      kind: "page",
      name: location.pathname,
      ts: Date.now(),
      sessionId: sessionId(),
      anonymousId: anonymousId(this.persist),
      path: location.pathname,
      ...(first && document.referrer ? { referrer: document.referrer } : {}),
      ...(this.userId ? { userId: this.userId } : {}),
    });
  }

  identify(userId: string): void {
    this.userId = userId;
    this.emit("identify", "identify");
  }

  destroy(): void {
    this.queue.flush();
    for (const fn of this.cleanup) fn();
  }
}

function watchHistory(onChange: () => void): () => void {
  const { pushState, replaceState } = history;
  history.pushState = function (...args: Parameters<History["pushState"]>) {
    pushState.apply(this, args);
    onChange();
  };
  history.replaceState = function (...args: Parameters<History["replaceState"]>) {
    replaceState.apply(this, args);
    onChange();
  };
  addEventListener("popstate", onChange);
  return () => {
    history.pushState = pushState;
    history.replaceState = replaceState;
    removeEventListener("popstate", onChange);
  };
}

  client = new Client(options);
}

  client?.emit("track", name, props);
}

export function identify(userId: string): void {
  client?.identify(userId);
}

export function _reset(): void {
  client?.destroy();
  client = undefined;
}
